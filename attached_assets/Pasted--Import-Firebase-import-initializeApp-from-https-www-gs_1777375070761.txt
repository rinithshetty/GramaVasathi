// Import Firebase
import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js";
import { getFirestore, collection, getDocs, addDoc } 
from "https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore.js";

// 🔥 Replace with your Firebase config
const firebaseConfig = {
  apiKey: "YOUR_KEY",
  authDomain: "YOUR_DOMAIN",
  projectId: "YOUR_ID",
};

const app = initializeApp(firebaseConfig);
const db = getFirestore(app);

let selectedHomestay = null;

// Load homestays
async function loadHomestays() {
  const querySnapshot = await getDocs(collection(db, "homestays"));
  const list = document.getElementById("homestay-list");

  querySnapshot.forEach((doc) => {
    const data = doc.data();

    const div = document.createElement("div");
    div.className = "card";
    div.innerHTML = `
      <h3>${data.name}</h3>
      <p>₹${data.price}</p>
      <p>⭐ ${data.rating}</p>
    `;

    div.onclick = () => showDetails(data);

    list.appendChild(div);
  });
}

// Show details
function showDetails(data) {
  selectedHomestay = data;

  document.getElementById("details").style.display = "block";
  document.getElementById("name").innerText = data.name;
  document.getElementById("price").innerText = "Price: ₹" + data.price;
  document.getElementById("activities").innerText =
    "Activities: " + data.activities.join(", ");
}

// Booking
window.bookNow = async function () {
  const name = document.getElementById("username").value;
  const date = document.getElementById("date").value;

  if (!name || !date) {
    alert("Please fill all fields");
    return;
  }

  await addDoc(collection(db, "bookings"), {
    name,
    date,
    homestay: selectedHomestay.name,
  });

  alert("Booking Confirmed!");
};

// Checklist score
window.calculateScore = function () {
  const checks = document.querySelectorAll(".check");
  let checked = 0;

  checks.forEach((c) => {
    if (c.checked) checked++;
  });

  const score = (checked / checks.length) * 100;

  document.getElementById("score").innerText =
    "Readiness Score: " + score + "%";
};

// Run
loadHomestays();
